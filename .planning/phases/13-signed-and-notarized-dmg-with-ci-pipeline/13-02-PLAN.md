Create `.github/workflows/release.yml` with this exact content:

```yaml
name: Automation Health Release

on:
  push:
    tags:
      - 'v*'
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: false

jobs:
  release:
    name: Build, sign, notarize, and release
    runs-on: macos-latest
    timeout-minutes: 60
    env:
      TZ: Europe/Berlin
    steps:
      - name: Check out repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Show toolchain
        run: swift --version

      - name: Import Apple Developer certificate
        env:
          APPLE_DEVELOPER_CERTIFICATE_P12: ${{ secrets.APPLE_DEVELOPER_CERTIFICATE_P12 }}
          APPLE_DEVELOPER_CERTIFICATE_PASSWORD: ${{ secrets.APPLE_DEVELOPER_CERTIFICATE_PASSWORD }}
        run: |
          if [ -z "$APPLE_DEVELOPER_CERTIFICATE_P12" ]; then
            echo "APPLE_DEVELOPER_CERTIFICATE_P12 secret is not set"
            echo "Generate it: base64 -i ~/Downloads/developerID_application.p12 | tr -d '\n' | pbcopy"
            exit 1
          fi
          echo "$APPLE_DEVELOPER_CERTIFICATE_P12" | base64 --decode > certificate.p12
          security create-keychain -p actions build.keychain
          security list-keychains -d user -s build.keychain login.keychain
          security default-keychain -s build.keychain
          security unlock-keychain -p actions build.keychain
          security import certificate.p12 -k build.keychain -P "$APPLE_DEVELOPER_CERTIFICATE_PASSWORD" -T /usr/bin/codesign -T /usr/bin/security
          security set-key-partition-list -S apple-tool:,apple: -s -k actions build.keychain
          security find-identity -v build.keychain
          rm certificate.p12

      - name: Import Developer ID identity name
        run: |
          DEVELOPER_IDENTITY=$(security find-identity -v -p codesigning build.keychain | grep "Developer ID Application" | head -1 | sed -E 's/.*"([^"]+)".*/\1/')
          echo "APPLE_DEVELOPER_IDENTITY=$DEVELOPER_IDENTITY" >> $GITHUB_ENV
          echo "Found identity: $DEVELOPER_IDENTITY"

      - name: Write App Store Connect API key
        env:
          APPLE_NOTARY_KEY: ${{ secrets.APPLE_NOTARY_KEY }}
        run: |
          if [ -z "$APPLE_NOTARY_KEY" ]; then
            echo "APPLE_NOTARY_KEY secret is not set"
            echo "Create an App Store Connect API Key at: https://appstoreconnect.apple.com/access/integrations/api"
            exit 1
          fi
          mkdir -p ~/.appstoreconnect
          echo "$APPLE_NOTARY_KEY" > ~/.appstoreconnect/AuthKey.p8
          chmod 600 ~/.appstoreconnect/AuthKey.p8
          echo "APPLE_NOTARY_KEY_FILE=$HOME/.appstoreconnect/AuthKey.p8" >> $GITHUB_ENV

      - name: Set notarization credentials
        run: |
          echo "APPLE_DEVELOPER_TEAM_ID=${{ secrets.APPLE_DEVELOPER_TEAM_ID }}" >> $GITHUB_ENV
          echo "APPLE_NOTARY_KEY_ID=${{ secrets.APPLE_NOTARY_KEY_ID }}" >> $GITHUB_ENV
          echo "APPLE_NOTARY_KEY_ISSUER_ID=${{ secrets.APPLE_NOTARY_KEY_ISSUER_ID }}" >> $GITHUB_ENV
          # GITHUB_TOKEN is already set by Actions

      - name: Build, sign, notarize, and package DMG
        run: ./script/release.sh

      - name: Upload DMG artifact
        uses: actions/upload-artifact@v4
        with:
          name: automation-health-dmg
          path: dist/AutomationHealth-*.dmg
          retention-days: 30

      - name: Create GitHub Release
        env:
          GH_TOKEN: ${{ secrets.GITHUB_TOKEN }}
        run: |
          DMG_FILE=$(ls dist/AutomationHealth-*.dmg | head -1)
          VERSION="${GITHUB_REF_NAME#v}"
          gh release create "$GITHUB_REF_NAME" \
            --title "Automation Health v$VERSION" \
            --notes "## Automation Health v$VERSION

          ### Installation

          1. Download the DMG below
          2. Open the DMG
          3. Drag Automation Health to your Applications folder

          ### Verification

          This release is signed and notarized by Apple. macOS Gatekeeper will allow it to run without security warnings.

          ### Building from Source

          See [README.md](https://github.com/${{ github.repository }}/blob/${{ github.ref_name }}/README.md) for source build instructions." \
            --verify-tag \
            "$DMG_FILE"

      - name: Verify release assets
        run: |
          echo "Release created: ${{ github.ref_name }}"
          DMG_FILE=$(ls dist/AutomationHealth-*.dmg | head -1)
          echo "DMG: $DMG_FILE"
          ls -lh "$DMG_FILE"
          xcrun stapler validate "$DMG_FILE" && echo "Staple ticket present" || echo "Warning: Staple ticket not found"
```

The workflow:
- Triggers on tag pushes matching `v*` pattern (e.g., `v1.3.0`) and manual `workflow_dispatch`
- Timeout of 60 minutes (notarization can take 30+ minutes)
- Permissions set to `contents: write` (needed for `gh release create`)
- Concurrency group prevents overlapping release runs but `cancel-in-progress: false` (releases must not be cancelled mid-notarization)
- `fetch-depth: 0` for full git history (needed for tag-based version detection in release.sh)
- Imports the Developer ID certificate from a base64-encoded GitHub secret `APPLE_DEVELOPER_CERTIFICATE_P12` into a temporary keychain
- Creates a temporary `build.keychain` and imports the certificate, setting it as default with partition list access for codesign
- Writes the App Store Connect API key from `APPLE_NOTARY_KEY` secret to `~/.appstoreconnect/AuthKey.p8` with 600 permissions
- Sets all required env vars from GitHub Secrets: `APPLE_DEVELOPER_TEAM_ID`, `APPLE_NOTARY_KEY_ID`, `APPLE_NOTARY_KEY_ISSUER_ID`
- Runs `./script/release.sh` for the full build-sign-notarize-package pipeline
- Uploads the DMG as a workflow artifact with 30-day retention
- Creates a GitHub Release using `gh release create` with the DMG attached, the tag verified, and human-readable release notes
- Verifies the stapled notarization ticket with `xcrun stapler validate`

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| GitHub Secrets -> CI runner | Apple Developer certificate (P12), private key password, and notary API key are stored as GitHub Secrets and injected into the runner environment at workflow start |
| CI runner -> Apple notary service | App Store Connect API key written to temp file on runner, used by notarytool over HTTPS |
| CI runner -> GitHub Releases API | GITHUB_TOKEN (auto-generated) used by gh CLI to create release and upload DMG |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-07 | Information Disclosure | .github/workflows/release.yml — certificate import | mitigate | Developer ID certificate P12 is base64-encoded in a GitHub Secret (masked in logs). The temporary keychain is created with a password and deleted when the runner is destroyed. The P12 password is a separate secret. |
| T-13-08 | Information Disclosure | .github/workflows/release.yml — API key file | mitigate | App Store Connect API key written to `~/.appstoreconnect/AuthKey.p8` with `chmod 600`. The file exists only during the workflow run and is destroyed with the runner. The key is never echoed or logged. |
| T-13-09 | Tampering | .github/workflows/release.yml — tag verification | mitigate | `gh release create ... --verify-tag` ensures the release is created from the tag that triggered the workflow, not a modified tag. The workflow triggers only on `v*` tag pushes (not branches). |
| T-13-10 | Elevation of Privilege | .github/workflows/release.yml — keychain partition list | mitigate | `security set-key-partition-list -S apple-tool:,apple:` restricts keychain access to Apple-signed tools (codesign, security, etc.), preventing other processes from using the signing identity. |
| T-13-11 | Denial of Service | .github/workflows/release.yml — notarization timeout | mitigate | Workflow timeout set to 60 minutes (Apple's notarization service typically completes in 5-15 minutes). `notarytool submit --wait` handles polling. If the service is down, the workflow fails naturally. |
</threat_model>
